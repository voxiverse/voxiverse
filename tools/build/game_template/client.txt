// @Name@'s client part (the game: games/@id@): it keeps the points the server last told
// the player and shows them in a line across the top of the screen.

GAME :: engine.Game {
	info     = @id@.INFO,
	kinds    = @id@.KINDS,
	state    = View,
	received = received,
	hud      = hud,
	look     = LOOK,
	// seen  = seen, // the engine's events (a hit, a death, a wound), made felt: see below
}

// LOOK: the game's atmosphere (docs/games.md, "A game's look"). Every field's zero is the
// engine's own, so as it stands this is the sandbox's look: give the game an air of its own,
// and keep its creatures, pickups and paths easy to see.
LOOK :: engine.Look {
	// time      = 0.5, // a time of day that stands (0.25 sunrise .. 0.75 sunset; 0: the clock)
	// horizon   = {0.42, 0.60, 0.88}, // the sky at the horizon, and the fog's colour
	// sun_light = {1, 1, 1}, // full daylight's colour
	// fog       = 96, // blocks to a whole fog (0: the view's edge)
	// grade     = {tint = {1, 1, 1}, saturation = 0, contrast = 0, vignette = 0},
}

// View: what the session knows of the game.
View :: struct {
	points: @id@.Points, // the server's last word (zero until the first)
}

view :: proc(s: ^engine.Session) -> ^View {return engine.game_state(s, View)}

received :: proc(s: ^engine.Session, r: protocol.Received) {
	if points, ok := engine.game_value(s, r, @id@.Points); ok && points != nil {
		view(s).points = points^
	}
}

// hud: the player's points, a line across the top of the screen.
hud :: proc(s: ^engine.Session, ctx: ^ui.Context) {
	rest := ctx.screen
	engine.top_bar(ctx, &rest, @id@.tr(.Points_Line, view(s).points.points))
}
